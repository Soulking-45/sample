// 7) Salary greater than or equal to 10000
// and less than 20000

db.Employee.find({
    salary: { $gte: 10000, $lt: 20000 }
})


// 8) Employees staying in Thane or Mumbai
// and salary greater than 10000

db.Employee.find({
    city: { $in: ["Thane", "Mumbai"] },
    salary: { $gt: 10000 }
})


// 9) Employees whose salary is not equal to 10000
// and work in Developer or Tester department

db.Employee.find({
    department: { $in: ["Developer", "Tester"] },
    salary: { $ne: 10000 }
})
